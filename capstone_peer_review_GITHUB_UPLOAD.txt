from fastapi import FastAPI
from src.model import predict
from src.logging_config import get_logger

app = FastAPI(title="Capstone Prediction API")
logger = get_logger()

@app.get("/health")
def health():
    return {"status": "ok"}

@app.post("/predict/{country}")
def predict_country(country: str, features: list[float]):
    value = predict(features)
    logger.info("Prediction requested for country=%s", country)
    return {"country": country, "prediction": value}

@app.post("/predict/all")
def predict_all(features_by_country: dict[str, list[float]]):
    predictions = {c: predict(x) for c, x in features_by_country.items()}
    logger.info("Prediction requested for all countries")
    return {"predictions": predictions}

import logging

def get_logger(name="capstone"):
    logger = logging.getLogger(name)
    if not logger.handlers:
        handler = logging.StreamHandler()
        handler.setFormatter(logging.Formatter("%(asctime)s %(levelname)s %(name)s: %(message)s"))
        logger.addHandler(handler)
        logger.setLevel(logging.INFO)
    return logger

import numpy as np

def predict(features):
    values = np.asarray(features, dtype=float)
    return float(values.mean()) if values.size else 0.0

import time

def measure_prediction_time(function, *args, **kwargs):
    start = time.perf_counter()
    result = function(*args, **kwargs)
    elapsed = time.perf_counter() - start
    return result, elapsed

from fastapi.testclient import TestClient
from src.api import app

client = TestClient(app)

def test_health():
    r = client.get("/health")
    assert r.status_code == 200
    assert r.json()["status"] == "ok"

def test_country_prediction():
    r = client.post("/predict/India", json=[1, 2, 3])
    assert r.status_code == 200
    assert r.json()["country"] == "India"

def test_all_country_prediction():
    r = client.post("/predict/all", json={"India": [1, 2], "USA": [3, 4]})
    assert r.status_code == 200
    assert set(r.json()["predictions"]) == {"India", "USA"}
